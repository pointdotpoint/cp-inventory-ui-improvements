// FastInventory: the inventory screen opens without the 3D V preview and loads it
// asynchronously a moment later. Closing the screen before the delay runs out skips the
// preview entirely, so quick open/close never pays for it.
//
// Vanilla spawns the preview through SpawnLibraryItemController on the "paperDoll" widget
// (library item "previewPanel" in inventory_screen.inkwidget). That spawn has already
// happened by the time OnSetUserData runs, but the puppet itself has not been created yet,
// so discarding the preview widget there cancels it. We spawn a fresh one later.

@addField(gameuiInventoryGameController)
private let m_fiTimerWidget: ref<inkWidget>;

@addField(gameuiInventoryGameController)
private let m_fiTimer: ref<inkAnimProxy>;

@addMethod(gameuiInventoryGameController)
private func FastInventoryDelay() -> Float {
    return 0.75;
}

@wrapMethod(gameuiInventoryGameController)
protected cb func OnSetUserData(userData: ref<IScriptable>) -> Bool {
    let doll: ref<inkCompoundWidget> = inkWidgetRef.Get(this.m_paperDollWidget) as inkCompoundWidget;
    if IsDefined(doll) {
        doll.RemoveAllChildren();
    }

    wrappedMethod(userData);

    if IsDefined(doll) {
        this.FastInventoryStartTimer();
    }
}

@wrapMethod(gameuiInventoryGameController)
protected cb func OnUninitialize() -> Bool {
    if IsDefined(this.m_fiTimer) {
        this.m_fiTimer.UnregisterFromAllCallbacks(inkanimEventType.OnFinish);
        this.m_fiTimer.Stop();
    }
    this.m_fiTimer = null;
    this.m_fiTimerWidget = null;

    wrappedMethod();
}

// Menus run while the game is paused, so an ink animation on a dummy widget is the timer.
@addMethod(gameuiInventoryGameController)
private func FastInventoryStartTimer() -> Void {
    let timerWidget: ref<inkCanvas> = new inkCanvas();
    timerWidget.SetName(n"fastInventoryTimer");
    timerWidget.Reparent(this.GetRootCompoundWidget());

    let interpolator: ref<inkAnimTransparency> = new inkAnimTransparency();
    interpolator.SetDuration(this.FastInventoryDelay());
    interpolator.SetStartTransparency(1.0);
    interpolator.SetEndTransparency(1.0);
    let definition: ref<inkAnimDef> = new inkAnimDef();
    definition.AddInterpolator(interpolator);

    this.m_fiTimerWidget = timerWidget;
    this.m_fiTimer = timerWidget.PlayAnimation(definition);
    this.m_fiTimer.RegisterToCallback(inkanimEventType.OnFinish, this, n"OnFastInventoryTimer");
}

@addMethod(gameuiInventoryGameController)
protected cb func OnFastInventoryTimer(proxy: ref<inkAnimProxy>) -> Bool {
    this.m_fiTimer = null;
    this.AsyncSpawnFromLocal(inkWidgetRef.Get(this.m_paperDollWidget), n"previewPanel");
}
